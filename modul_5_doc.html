A. Operasi Sistem Pertemanan (server/pkg/database/repository.go)

// Mengirim permintaan pertemanan
func (r *Repository) SendFriendRequest(fromUserID, targetUsername string) (*Friendship, error) {
    targetUser, exists := r.usersByUname[targetUsername]
    if !exists { return nil, fmt.Errorf("user tidak ditemukan") }
    friendship := &Friendship{
        UserID:    fromUserID,
        FriendID:  targetUser.ID,
        Status:    "PENDING",
    }
    return friendship, nil
}
